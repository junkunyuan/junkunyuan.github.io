AnyFlow: Any-Step Video Diffusion Model with On-Policy Flow Map Distillation

Yuchao Gu (5,435), Guian Fang (245), Yuxin Jiang (278), Weijia Mao (1,369), Song Han (108,706), Han Cai (14,483), Mike Zheng Shou (24,141)

videodiffusionflowdistillationunread

It distills a video diffusion model into a flow map between any two times and then refines it on-policy by flow map backward simulation.
Insight. A student on the teacher's ODE keeps test-time scaling. Judge few-step students at all budgets, and train them as they sample.

  • Flow map (§3.1, Eq. 2). A network learns the exact transport of the PF-ODE, fθ(zt,t,r)≈zr for 1≥t>r≥0, and fθ(zt,t,t)=zt.

  • Central-difference MeanFlow target (§3.1, Eq. 3–4). With fθ=zt−(t−r)uθ(zt,r,t), the loss is ‖uθ−sg(utgt)‖22 and utgt=v−(t−r)duθ/dt Eq. (3). FSDP lacks a JVP, so two forward passes give (u(zt+ϵ,t+ϵ)−u(zt−ϵ,t−ϵ))/2ϵ Eq. (4), with zt±ϵ=zt±ϵv and ϵ=5/1000.

  • Time sampler and weight (§4.2.1). t>r come from U(0,1) with shift 5, half the batch at t=r, a quarter at r=0, and w(t)=Beta(2,1.5).

  • Fused guidance (Eq. 6). u=(uc−(1−g)sg(u∅))/g with the teacher's scale g=3 and the null-text prediction u∅. Sampling has no CFG.

  • Adaptive reweighting (Eq. 7). wt,r=μt=r/(‖Δ‖22+c) for t≠r and 1 at t=r, where Δ is the error and μt=r is the mean boundary loss.

  • Flow map backward simulation (Eq. 8). Because fθ(zt,t,q)≈fθ(fθ(zt,t,r),r,q), a rollout is three calls with gradient, T→t→r→0.

  • DMD loss (Eq. 5). ∇θℒ=−𝔼[(sreal−sfake)∂fθ/∂θ] on the noised z^0. The CFG-3 teacher gives sreal, and a LoRA copy gives sfake.

  • Experiments (§5). At 4 NFEs, the Wan2.1 1.3B student gets 83.48 on VBench, above rCM's 82.51 and above its 50×2-step teacher.

The training and sampling code. Follow Algorithm 1 and 2 of the paper.

def shift_t(t, shift=5.0):
    return shift * t / (1 + (shift - 1) * t)  # the teacher's timestep shift

def sample_time_steps(bs, diffusion_ratio=0.5, consistency_ratio=0.25):
    u1, u2 = torch.rand(bs), torch.rand(bs)
    t, r = torch.maximum(u1, u2), torch.minimum(u1, u2)  # make r <= t
    idx = torch.arange(bs)
    is_diffusion = idx < round(diffusion_ratio * bs)  # boundary rows, r = t
    r = torch.where(is_diffusion, t, r)
    r = torch.where(~is_diffusion & (idx < round((diffusion_ratio + consistency_ratio) * bs)), 0.0, r)  # r = 0
    return shift_t(t), shift_t(r), is_diffusion

def train_weight(t, n=1000):
    grid = shift_t(torch.linspace(1, 0, n + 1)[:-1])
    w = grid * (1 - grid) ** 0.5  # Beta(2, 1.5) shape
    w = w * n / w.sum()
    return w[(grid[:, None] - t.flatten()[None]).abs().argmin(0)]  # nearest grid timestep

def central_difference(model, z_t, v, c, t, r, guidance=3.0, epsilon=5, num_train_timesteps=1000):
    d = epsilon / num_train_timesteps
    with no_grad():
        return (model(z_t + d * v, c, t + d, r) - model(z_t - d * v, c, t - d, r)) / (2 * d * guidance)  # Eq. 4

def train(model, x, c, null, guidance=3.0, drop_text_ratio=0.1, adaptive_eps=1e-5):
    # 1. Sample the pair r <= t.
    t, r, is_diffusion = sample_time_steps(x.shape[0])

    # 2. Build the noisy latent z_t and its velocity v.
    noise = torch.randn_like(x)
    z_t = (1 - t) * x + t * noise  # linear path
    v = noise - x  # conditional velocity

    # 3. Predict the average velocity with a text dropout, and fuse the guidance into the prediction.
    c = where(rand(x.shape[0]) < drop_text_ratio, null, c)
    u_c = model(z_t, c, t, r)
    with no_grad():
        u_n = model(z_t, null, t, r)
    u = (u_c - (1 - guidance) * u_n) / guidance  # Eq. 6

    # 4. Build the target from a central difference along v.
    dudt = central_difference(model, z_t, v, c, t, r, guidance)
    u_tgt = v - (t - r) * dudt  # Eq. 3

    # 5. Calculate the weighted loss, and rescale the transitions to the boundary loss.
    loss = (u - u_tgt).square().flatten(1).mean(1) * train_weight(t)
    mu = loss[is_diffusion].mean().detach()
    loss = where(is_diffusion, loss, loss * mu / (loss.detach() + adaptive_eps))  # Eq. 7
    return loss.mean()

def rollout(model, noise, c, num_steps_list=(2, 4, 8, 16, 50)):
    n = choice(num_steps_list)
    j = randint(0, n)  # the step that carries the target transition
    ts = cat([shift_t(linspace(1, 0, n + 1)[:-1]), [0.0]])
    z = noise
    for a, b in ((ts[0], ts[j]), (ts[j], ts[j + 1]), (ts[j + 1], ts[-1])):  # Eq. 8
        if a != b:
            z = z - (a - b) * model(z, c, a, b)
    return z

def dmd_grad(teacher, fake, z0, c, null, real_guidance_scale=3.0):
    with no_grad():
        t = shift_t(rand(z0.shape[0]))
        z_t = (1 - t) * z0 + t * torch.randn_like(z0)
        x_fake = z_t - t * fake(z_t, c, t, t)
        x_c, x_n = z_t - t * teacher(z_t, c, t, t), z_t - t * teacher(z_t, null, t, t)
        x_real = x_c + real_guidance_scale * (x_c - x_n)
        grad = (x_fake - x_real) / (z0 - x_real).abs().mean(dim=(1, 2, 3, 4), keepdim=True)  # Eq. 5
    return torch.nan_to_num(grad)

def distill(model, teacher, fake, x, c, null, dmd_weight=1.0):
    # 1. Roll out the student by flow map backward simulation, with the gradient through all three calls.
    z0 = rollout(model, torch.randn_like(x), c)

    # 2. Calculate the DMD loss on the re-noised rollout.
    grad = dmd_grad(teacher, fake, z0, c, null)
    loss = dmd_weight * (z0 - (z0 - grad).detach()).square().mean()

    # 3. Add the forward loss.
    loss = loss + train(model, x, c, null)

    # 4. Refit the fake score to fresh rollouts by flow matching.
    with no_grad():
        z0 = rollout(model, torch.randn_like(x), c)
    t = shift_t(torch.sigmoid(torch.randn(x.shape[0])))  # logit-normal(0, 1)
    noise = torch.randn_like(z0)
    loss_fake = (fake((1 - t) * z0 + t * noise, c, t, t) - (noise - z0)).square().mean()

def sample(model, c, shape, num_steps=4):
    ts = cat([shift_t(linspace(1, 0, num_steps + 1)[:-1]), [0.0]])
    z = torch.randn(shape)
    for t, r in zip(ts[:-1], ts[1:]):
        z = z - (t - r) * model(z, c, t, r)  # Euler on the flow map, no CFG
    return z

videoflowdistillationunread

A main backbone runs once per transition and a recurrent flow head makes cheap inner steps, with MeanFlow pretraining and DMD2.
Insight. A step can be less than a full network call. Split semantics from refinement, and let distillation gradients reach the inner steps.

  • Transition matching (§2, Eq. 1–5). On xt=(1−t)x+tx1 Eq. (1), with data x and noise x1, flow matching needs many small steps. TM jumps xti−1=xti−(ti−ti−1)y Eq. (4) with a latent y:=x1−x Eq. (3), which an inner flow ys=(1−s)y+sy1 Eq. (5) in a light head samples.

  • MeanFlow objective (§2, Eq. 7–10). A flow map f(ys,s,r)=ys−∫srvdτ Eq. (7) uses an average velocity u. Differentiation gives u+(s−r)du/ds=v Eq. (8), so the loss is ‖uθ−sg(v−(s−r)duθ/ds)‖2 Eq. (9–10), where v≈y1−y is the conditional inner velocity.

  • Decoupled student (Eq. 12). The first L−H blocks are the backbone mθ, and the last H=5 are the head ysj−1≈fθ(ysj,sj,sj−1;mθ).

  • Stage 1, TM-MF (§3.1, Eq. 13–14). fθ(ys,s,r;m):=ys+(s−r)uθ(ys,s,r;m) Eq. (13) and uθ:=y1−headθ(ys,s,r;m) Eq. (14). A central difference with ys±δ=ys±δv and δ=0.005 replaces the JVP, one-sided at the ends, as FSDP and flash attention lack forward-mode AD.

  • Inner times, guidance (App. A.2). s is uniform with shift 3, r is the grid point below, and 75% of batches have r=s. v uses CFG 3.

  • DMD2-v (Eq. 11). The loss is 𝔼[w(t)sg(D(x^t,t))⊤x^], where w(t) is the DMD weighting, sg is the stop-gradient, x^t is the noised student sample x^, and D is the fake score minus the teacher score (CFG 5). A Conv3D GAN head on teacher features adds a GAN loss with λ=0.03.

  • Timestep shifting (§3.2). t=γt′/((γ−1)t′+1) on uniform t′, with γ=5 for DMD noise and 10 for student grids, against mode collapse.

  • Flow head rollout (§3.2, Eq. 15). gθ(xti,ti;y1):=x1−InnerFlow(mθ(xti,ti)) unrolls the head into a generator. The VSD gradient goes through all inner steps, which is backward simulation of the inner flow. This costs under 17% more compute at N=2 with 5 of 30 blocks.

  • Effective NFE (§4.1, Eq. 16). NFE:=M(1+(N−1)H/L) for M transitions, each with one full pass and N−1 passes of the H-block head.

  • Experiments (§4). VBench: Wan2.1 1.3B N2H5 gets 84.68 at 2.33 NFE (rCM 84.43 at 4), and 14B gets 84.24 at 1.38 (rCM 83.02).

  • Fractional steps lose at two outer steps on 14B (Tab. 2). N4H5 at 2.75 NFE gets 84.62, below the 85.05 of two-step rCM.

def shift(t, gamma):
    return gamma * t / ((gamma - 1) * t + 1)  # timestep shifting

def student_grid(n, gamma=10):
    return shift(linspace(1, 0, n + 1), gamma).clamp(max=0.999)  # t_M = s_N = 0.999 as in Wan

def sample_inner_times(bs, grid, shift_mf=3, ratio_r_equal_s=0.75):
    s = shift(torch.rand(bs).clamp(0.001, 0.999), shift_mf)
    r = grid[torch.searchsorted(-grid, -s)]  # the grid point just below s
    r = torch.where(torch.rand(bs) < ratio_r_equal_s, s, r)
    return s, r

def inner_flow(head, m, grid):
    y_1 = torch.randn(shape)
    y = y_1
    for s, r in zip(grid[:-1], grid[1:]):
        y = y - (s - r) * (y_1 - head(y, s, r, m))  # Eq. 13 with Eq. 7's sign, Eq. 14
    return y

def adaptive_loss(error, c_per_dim=1.0):
    loss = error.square().flatten(1).sum(dim=1)
    return loss / (loss.detach() + c_per_dim * error[0].numel())  # c = d (1.3B) or 1e-5 d (14B)

def pretrain(backbone, head, x, c, num_outer=2, num_inner=4, cfg_scale=3.0, cond_dropout=0.1, delta=0.005):
    # 1. Add noise to the video at an outer grid time and run the backbone once.
    t_i = student_grid(num_outer)[torch.randint(num_outer, (bs,))]
    x_1 = torch.randn_like(x)
    x_t = (1 - t_i) * x + t_i * x_1  # Eq. 1
    dropped = rand(bs) < cond_dropout
    m = backbone(x_t, t_i, where(dropped, null, c))  # never detached

    # 2. Build the inner target y and its noisy state at inner times s and r.
    y = x_1 - x  # Eq. 3
    y_1 = torch.randn_like(y)
    s, r = sample_inner_times(bs, student_grid(num_inner))
    y_s = (1 - s) * y + s * y_1  # Eq. 5

    # 3. Guide the conditional inner velocity with the head's unconditional prediction.
    with no_grad():
        u_null = y_1 - head(y_s, s, s, backbone(x_t, t_i, null))
    v = where(dropped, y_1 - y, cfg_scale * (y_1 - y) + (1 - cfg_scale) * u_null)

    # 4. Predict the average velocity and its target, with a central difference for the JVP.
    u = y_1 - head(y_s, s, r, m)  # Eq. 14
    with no_grad():
        u_fn = lambda y_, s_: y_1 - head(y_, s_, r, m)
        dudt = (u_fn(y_s + delta * v, s + delta) - u_fn(y_s - delta * v, s - delta)) / (2 * delta)
        u_tgt = v - (s - r) * dudt  # Eq. 10

    # 5. Calculate the normalized loss.
    loss = adaptive_loss(u - u_tgt).mean()  # Eq. 9

def distill(
    backbone, head, teacher, fake_score, disc, x, c,
    num_outer=2, num_inner=4, cfg_scale=5.0, gan_weight=0.03
):
    # 1. Add noise to the video at an outer grid time and run the backbone once.
    t_i = student_grid(num_outer)[torch.randint(num_outer, (bs,))]
    x_1 = torch.randn_like(x)
    m = backbone((1 - t_i) * x + t_i * x_1, t_i, c)

    # 2. Unroll the flow head into a generator, with the gradient through every inner step.
    x_hat = x_1 - inner_flow(head, m, student_grid(num_inner))  # Eq. 15

    # 3. Add noise to the sample and take the direction from the teacher's score to the fake one.
    t = shift(rand(bs).clamp(0.001, 0.999), 5)
    x_hat_t = (1 - t) * x_hat + t * torch.randn_like(x_hat)
    with no_grad():
        x_fake = x_hat_t - t * fake_score(x_hat_t, t, c)  # Eq. 17
        v_c, v_u = teacher(x_hat_t, t, c), teacher(x_hat_t, t, null)
        x_real = x_hat_t - t * (v_u + cfg_scale * (v_c - v_u))  # Eq. 17
        grad = (x_fake - x_real) / abs(x_hat - x_real).mean()  # DMD weighting w(t)

    # 4. Calculate the VSD loss plus the GAN loss on teacher features.
    loss = (x_hat * grad).mean() + gan_weight * softplus(-disc(teacher.features(x_hat_t, t, c))).mean()  # Eq. 11

def critic_step(fake_score, disc, teacher, x_hat, x, c):
    # 1. Refit the fake score on the student's samples by flow matching.
    t = shift(rand(bs).clamp(0.001, 0.999), 5)
    noise = torch.randn_like(x)
    x_hat_t = (1 - t) * x_hat.detach() + t * noise
    loss_fake = (fake_score(x_hat_t, t, c) - (noise - x_hat.detach())).square().mean()

    # 2. Train the Conv3D discriminator on teacher features of real videos and samples.
    x_t = (1 - t) * x + t * noise
    loss_disc = softplus(disc(teacher.features(x_hat_t, t, c))).mean()
    loss_disc = loss_disc + softplus(-disc(teacher.features(x_t, t, c))).mean()

def sample(backbone, head, c, num_outer=2, num_inner=4):
    x = torch.randn(shape)
    ts = student_grid(num_outer)
    for t_i, t_prev in zip(ts[:-1], ts[1:]):
        m = backbone(x, t_i, c)  # one full pass, then N head passes: Eq. 16
        x = x - (t_i - t_prev) * inner_flow(head, m, student_grid(num_inner))  # Eq. 4

Between its few network calls, the student integrates the ODE with a network-free policy and learns by on-policy imitation of the teacher.
Insight. Network calls and integration steps can differ. A closed-form policy between the calls turns distillation into plain velocity imitation.

  • Policy-based flow (§2–3, Eq. 2). A teacher Gθ(xt,t) predicts u=(xt−x0)/t on xt=(1−t)x0+tϵ with one call for each velocity. The student π←Gϕ(xtsrc,tsrc) gives a network-free policy once per step, and xtdst=xtsrc+∫πdt uses Euler substeps of 1/128 at negligible cost.

  • GMFlow policy (§3.2). The velocity is a Gaussian mixture q(u∣xtsrc)=∏i∑k=1KAik𝒩(ui;μik,s2I) that adapts to the current xt.

  • Closed-form velocity (App. F, Eq. 21–28). Let μxk=xtsrc−σtsrcμk, sx=σtsrcs Eq. (21), and ν, ζ be the SNR-weighted gaps between xt and xtsrc: μk′=(sx2ν+μxk)/(sx2ζ+1) Eq. (25), Eq. (27) gives logits ak′, and with their softmax Ak′, π(xt,t)=(xt−∑kAk′μk′)/t Eq. (28).

  • π-ID, on-policy imitation (§4, Alg. 1). The detached policy πD rolls out to xt=xtsrc+∫tsrctπDdt, t~U(tdst,tsrc), and the loss is 12‖Gθ(xt,t,c)−π(xt,t)‖2. The teacher Gθ corrects states that the student itself visits, so the error grows linearly, not quadratically.

  • Starting states (Alg. 2, App. B). Data gets noise at a grid time shifted by t=mτ/(1+(m−1)τ), m=3.2, with the last step halved.

  • GM dropout and temperature (§5.1, App. B.1). Training drops components of πD at p=0.05 (0.1 at 1 MP) to explore new modes. Inference sharpens the mixture to weights Ak1/T and variance s2T Eq. (4) with T=0.3 at 4 NFE, but not on the last step, to keep fine textures.

  • Micro-window matching (Eq. 6). The target is the window-mean policy velocity, 12‖Gθ(xt,t)−1Δt∫t−Δttπdt‖2, with Δt=3/128.

  • Scheduled trajectory mixing (App. B.2). On FLUX.1, rollouts start as teacher paths, whose share falls linearly to zero in 2K steps.

  • Experiments (§5). 1-NFE FID is 2.85 on ImageNet-256 (MeanFlow 3.43), and 4-NFE FID is 14.3 on FLUX.1 (SenseFlow 23.3).

  • Fewer teacher steps, sharper images (Fig. 9). FLUX.1 dev is blurrier at 128 NFE than at 43, so the window also sets sharpness.

def warp(tau, shift=3.2):
    return shift * tau / (1 + (shift - 1) * tau)  # time shifting

def sample_segment(bs, nfe=4, final_step_scale=0.5):
    base = 1 / (nfe - 1 + final_step_scale)
    idx = torch.ceil(torch.rand(bs).clamp(min=1e-4) / base + 1 - final_step_scale).clamp(1, nfe)
    tau_src = ((idx - 1 + final_step_scale) * base).clamp(max=1)
    return tau_src, tau_src - torch.where(idx == 1, final_step_scale * base, base)  # last step halved

def gm_policy(means, logstds, logweights, x_src, t_src):
    return x_src - t_src * means, (2 * logstds).exp() * t_src ** 2, logweights, x_src, t_src  # Eq. 21

def gm_dropout(gm, p=0.1):
    mu_x, var_x, logweights, x_src, t_src = gm
    drop = torch.rand(logweights.shape[:2] + (1, 1, 1)) < p  # whole components
    drop &= ~drop.all(dim=1, keepdim=True)  # never all of them
    return mu_x, var_x, logweights.masked_fill(drop, float('-inf')), x_src, t_src

def gm_temperature(gm, temperature=0.3):
    mu_x, var_x, logweights, x_src, t_src = gm
    return mu_x, var_x * temperature, torch.log_softmax(logweights / temperature, dim=1), x_src, t_src  # Eq. 4

def gm_velocity(gm, x_t, t):
    mu_x, var_x, logweights, x_src, t_src = gm
    nu = (1 - t) * x_t / t ** 2 - (1 - t_src) * x_src / t_src ** 2  # below Eq. 22
    zeta = ((1 - t) / t) ** 2 - ((1 - t_src) / t_src) ** 2
    mu = (var_x * nu + mu_x) / (var_x * zeta + 1)  # Eq. 25
    logits = logweights + (mu_x * (nu - 0.5 * zeta * mu_x)).sum(dim=2, keepdim=True) / (var_x * zeta + 1)  # Eq. 27
    x_0 = (torch.softmax(logits, dim=1) * mu).sum(dim=1)  # Eq. 26
    return (x_t - x_0) / t  # Eq. 28

def integrate(gm, x, tau_start, tau_end, total_substeps=128):
    n = ((tau_start - tau_end) * total_substeps).ceil().clamp(min=1)
    dtau, tau, t = (tau_start - tau_end) / n, tau_start, warp(tau_start)
    for i in range(int(n.max())):  # Euler substeps, no network
        t_next = warp((tau - dtau).clamp(min=0))
        x_next = x + gm_velocity(gm, x, t) * (t_next - t)
        active = n > i
        x, t, tau = where(active, x_next, x), where(active, t_next, t), where(active, tau - dtau, tau)
    return x, t

def train(
    model, teacher, x_0, c, iteration,
    policy_dropout=0.1, num_states=4, window=3 / 128, teacher_cfg=3.5, num_decay_iters=2000
):
    # 1. Pick a segment of the student grid and add noise to the data at its shifted start.
    tau_src, tau_dst = sample_segment(bs)
    t_src = warp(tau_src)
    x_src = (1 - t_src) * x_0 + t_src * torch.randn_like(x_0)

    # 2. Get the policy with one student call, and a detached copy with dropout for the rollout.
    gm = gm_policy(*model(x_src, t_src, c), x_src, t_src)
    gm_d = gm_dropout(tuple(g.detach() for g in gm), policy_dropout)

    # 3. Split the segment into policy and teacher stretches, and decay the teacher share to zero.
    teacher_ratio = 1 - min(iteration, num_decay_iters) / num_decay_iters
    span = tau_src - tau_dst - window
    cuts = torch.sort(torch.rand(bs, num_states)).values * ((1 - teacher_ratio) * span)[:, None]
    gaps_pi = torch.diff(cuts, prepend=torch.zeros(bs, 1))
    cuts = torch.sort(torch.rand(bs, num_states - 1)).values
    gaps_teacher = torch.diff(cuts, prepend=torch.zeros(bs, 1), append=torch.ones(bs, 1))
    gaps_teacher = gaps_teacher * (teacher_ratio * span)[:, None]

    # 4. At each visited state, match the policy's window velocity to the teacher's.
    loss, x, tau = 0, x_src, tau_src
    for k in range(num_states):
        tau_a = (tau - gaps_pi[:, k]).clamp(min=0)
        tau_b = (tau_a - gaps_teacher[:, k]).clamp(min=0)
        with no_grad():
            x_a, t_a = integrate(gm_d, x, tau, tau_a)  # rollout of the detached policy
            u_teacher = teacher(x_a, t_a, c, guidance=teacher_cfg)
        x_w, t_w = integrate(gm, x_a, tau_a, tau_b - window)
        u_policy = (x_a - x_w) / (t_a - t_w)  # Eq. 6
        loss = loss + 0.5 * (u_teacher - u_policy).square().mean() / num_states
        x, tau = x_a + u_teacher * (warp(tau_b) - t_a), tau_b  # teacher stretch

def time_G(bs, p_mean=0.7, p_std=1.6):
    return atan(torch.exp(torch.randn(bs, 1, 1, 1, 1) * p_std + p_mean))  # lognormal sigma, in TrigFlow time

def time_D(bs, shift=5.0):
    u = torch.rand(bs, 1, 1, 1, 1)
    rf_t = shift * u / (1 + (shift - 1) * u)  # the teacher's timestep shift
    return atan(rf_t / (1 - rf_t))

def guided(teacher, x_t, t, c, null, teacher_guidance=5.0):
    (x0_c, F_c), (x0_u, F_u) = teacher(x_t, t, c), teacher(x_t, t, null)  # each net returns (x0, F)
    return x0_u + teacher_guidance * (x0_c - x0_u), F_u + teacher_guidance * (F_c - F_u)

def rollout(student, shape, c, n_steps, with_grad):
    ts = [full((shape[0], 1, 1, 1, 1), pi / 2)]
    for _ in range(n_steps - 1):
        ts.append(torch.minimum(time_D(shape[0]), ts[-1]))
    ts.append(0 * ts[-1])
    x = torch.randn(shape)
    for i, (t_cur, t_next) in enumerate(zip(ts[:-1], ts[1:])):
        with (enable_grad if with_grad and i == n_steps - 1 else no_grad)():
            x, _ = student(x, t_cur, c)  # gradient through the last denoise only
        x = cos(t_next) * x + sin(t_next) * torch.randn_like(x)  # re-noise, and t_next = 0 keeps x
    return x

def scm_loss(student, teacher, x0, c, null, it, tangent_warmup=1000, loss_scale=100.0):
    t = time_G(x0.shape[0])
    x_t = cos(t) * x0 + sin(t) * torch.randn_like(x0)
    with no_grad():
        _, F_teacher = guided(teacher, x_t, t, c, null)
        tangents = (cos(t) * sin(t) * F_teacher, cos(t) * sin(t))  # JVP rearrangement
        _, dF = jvp(lambda x, t_: student(x, t_, c)[1], (x_t, t), tangents)
    _, F_s = student(x_t, t, c)
    F_sg = F_s.detach()
    r = min(1.0, it / tangent_warmup)  # tangent warmup
    g = -cos(t) * sqrt(1 - r ** 2 * sin(t) ** 2) * (F_sg - F_teacher) - (r * cos(t) * sin(t) * x_t + dF)  # Eq. 5
    g = torch.nan_to_num(g)  # a BF16 overflow zeroes its entries
    g = g / (g.norm(dim=(1, 2, 3, 4), keepdim=True) + 0.1)
    return loss_scale * (F_s - F_sg - g).square().sum(dim=(1, 2, 3, 4))  # Eq. 4

def dmd_loss(student, teacher, fake, x0, c, null, n_steps, loss_scale_dmd=1.0):
    x0_s = rollout(student, x0.shape, c, n_steps, with_grad=True)
    t = time_D(x0.shape[0])
    x_t = cos(t) * x0_s + sin(t) * torch.randn_like(x0_s)
    with no_grad():
        x0_fake, _ = fake(x_t, t, c)
        x0_teacher, _ = guided(teacher, x_t, t, c, null)
        weight = (x0_s - x0_teacher).abs().mean(dim=(1, 2, 3, 4), keepdim=True).clip(min=1e-5)
    grad = (x0_fake - x0_teacher) / weight
    return loss_scale_dmd * (x0_s - (x0_s - grad).detach()).square().sum(dim=(1, 2, 3, 4))  # Eq. 6

def critic_loss(student, fake, x0, c, n_steps):
    x0_s = rollout(student, x0.shape, c, n_steps, with_grad=False)
    t = time_D(x0.shape[0])
    x_t = cos(t) * x0_s + sin(t) * torch.randn_like(x0_s)
    x0_fake, _ = fake(x_t, t, c)
    return ((x0_s - x0_fake).square() / sin(t) ** 2).sum(dim=(1, 2, 3, 4))  # flow matching on student samples

def train(student, teacher, fake, x0, c, null, it, tangent_warmup=1000, student_update_freq=5, max_steps=4):
    # 1. Pick the phase. All warmup iterations are student steps, and after the warmup, one in student_update_freq.
    student_phase = it < tangent_warmup or (it - tangent_warmup) % student_update_freq == 0
    it_s = it if it < tangent_warmup else tangent_warmup + (it - tangent_warmup) // student_update_freq

    # 2. Update the student with the sCM loss, and add the DMD loss after the warmup.
    if student_phase:
        loss = scm_loss(student, teacher, x0, c, null, it, tangent_warmup)
        if it >= tangent_warmup:
            loss = loss + dmd_loss(student, teacher, fake, x0, c, null, it_s % max_steps + 1)  # lambda = 1 / 100

    # 3. Otherwise, update the fake score on fresh student samples.
    else:
        loss = critic_loss(student, fake, x0, c, (it - it_s - 1) % max_steps + 1)

def add_noise(x_0, noise, t):
    return (1 - t / 1000) * x_0 + (t / 1000) * noise  # Eq. 1 with alpha = 1 - sigma

def shift(t, timestep_shift=8.0, min_step=20, max_step=980):
    t = timestep_shift * (t / 1000) / (1 + (timestep_shift - 1) * (t / 1000)) * 1000
    return t.clamp(min_step, max_step)

def block_causal_mask(num_frames, frames_per_block=3):
    i = torch.arange(num_frames)
    return (i[None, :] // frames_per_block) <= (i[:, None] // frames_per_block)  # Eq. 5

def sample_block_indices(bs, num_frames, num_steps, frames_per_block=3):
    idx = torch.randint(num_steps, (bs, num_frames // frames_per_block))  # one step per chunk
    return idx.repeat_interleave(frames_per_block, dim=1)

def init(student, ode_latents, c, steps=(1000, 757, 522, 0)):
    # 1. Read the teacher's ODE trajectory at one student step per chunk.
    idx = sample_block_indices(bs, num_frames, len(steps))
    t = torch.tensor(steps)[idx]
    x_t = ode_latents[torch.arange(bs)[:, None], idx, torch.arange(num_frames)[None]]
    x_0 = ode_latents[:, -1]  # the trajectory's clean end

    # 2. Regress the clean frames from the noisy ones.
    x_hat = student(x_t, t, c, mask=block_causal_mask(num_frames))
    loss = (x_hat - x_0)[t != 0].square().mean()  # Eq. 6

def train(student, teacher, fake_score, x_0, c, steps=(1000, 757, 522, 0), guidance_scale=3.5):
    # 1. Add noise to each chunk of a real video at its own step, and predict the clean frames.
    t_b = torch.tensor(steps)[sample_block_indices(bs, num_frames, len(steps))]
    x_t = add_noise(x_0, torch.randn_like(x_0), t_b)
    x_hat = student(x_t, t_b, c, mask=block_causal_mask(num_frames))

    # 2. Add noise to the prediction at one shifted step, and score it with the fake and guided teacher scores.
    with no_grad():
        t = shift(torch.randint(1000, (bs, 1)))  # one step for the whole video
        x_hat_t = add_noise(x_hat, torch.randn_like(x_hat), t)
        x_fake = fake_score(x_hat_t, t, c)
        x_c, x_u = teacher(x_hat_t, t, c), teacher(x_hat_t, t, null)  # bidirectional, frozen
        x_real = x_c + guidance_scale * (x_c - x_u)
        grad = (x_fake - x_real) / abs(x_hat - x_real).mean(dim=(1, 2, 3, 4), keepdim=True)  # Eq. 4

    # 3. Calculate a loss whose gradient on the prediction is the DMD gradient.
    loss = 0.5 * (x_hat.double() - (x_hat - grad).double().detach()).square().mean()

def critic_step(student, fake_score, x_0, c, steps=(1000, 757, 522, 0)):

    with no_grad():
        t_b = torch.tensor(steps)[sample_block_indices(bs, num_frames, len(steps))]
        x_gen = student(add_noise(x_0, torch.randn_like(x_0), t_b), t_b, c, mask=block_causal_mask(num_frames))

    # 2. Fit the fake score to them by flow matching on fresh noise.
    t = shift(torch.randint(1000, (bs, 1)))

def karras_sigmas(n=1000, sigma_min=0.002, sigma_max=80.0, rho=7.0):
    ramp = torch.linspace(0, 1, n)
    sigmas = (sigma_max ** (1 / rho) + ramp * (sigma_min ** (1 / rho) - sigma_max ** (1 / rho))) ** rho
    return sigmas.flip(0)  # small sigma first

def dm_loss(x, labels, real_unet, fake_unet, sigmas, min_step=20, max_step=980):
    with no_grad():
        sigma = sigmas[torch.randint(min_step, max_step + 1, (x.shape[0], 1, 1, 1))]  # t in [0.02T, 0.98T]
        noisy = x + sigma * torch.randn_like(x)  # Eq. 3
        p_real = x - real_unet(noisy, sigma, labels)  # Eq. 4, the frozen teacher
        p_fake = x - fake_unet(noisy, sigma, labels)  # Eq. 5
        weight = p_real.abs().mean([1, 2, 3], keepdim=True)  # Eq. 8
        grad = torch.nan_to_num((p_real - p_fake) / weight)  # Eq. 7
    return 0.5 * (x - (x - grad).detach()).square().mean()  # a surrogate loss whose gradient is grad

def fake_loss(x, labels, fake_unet, sigmas, sigma_data=0.5):
    x = x.detach()  # no gradient to the generator
    sigma = sigmas[torch.randint(0, len(sigmas), (x.shape[0], 1, 1, 1))]
    pred = fake_unet(x + sigma * torch.randn_like(x), sigma, labels)
    weight = sigma ** -2 + 1 / sigma_data ** 2  # Karras weighting
    return (weight * (pred - x) ** 2).mean()  # Eq. 6

def train(generator, real_unet, fake_unet, labels, z_ref, y_ref, lambda_reg=0.25, conditioning_sigma=80.0):
    # 1. Generate images in one step, from fresh noise and from the paired noise.
    sigmas = karras_sigmas()
    G = lambda z: generator(z * conditioning_sigma, conditioning_sigma, labels)
    x, x_ref = G(torch.randn_like(z_ref)), G(z_ref)

    # 2. Calculate the generator loss: distribution matching plus regression.
    loss_g = dm_loss(x, labels, real_unet, fake_unet, sigmas) + lambda_reg * lpips(x_ref, y_ref)  # Eq. 9

    # 3. Calculate the fake-score loss on the generator's own samples.
    loss_fake = fake_loss(x, labels, fake_unet, sigmas)
    return loss_g, loss_fake  # one update of each per step

  • Target (§3, Alg. 2). The teacher takes two DDIM steps t=i/N→t−0.5/N→t″=t−1/N, and the target x~=(zt″−(σt″/σt)zt)/(αt″−(σt″/σt)αt) inverts one student step. Unlike x, zt fully determines x~, so the student predicts it sharply.

  • Repeated halving (§3). A converged student is the next teacher, and N halves, 8192 to 1 on CIFAR-10, 50K updates per round.

  • Why ε-prediction fails (§4, Eq. 9). At zero SNR, x^=(zt−σtϵ^)/αt divides by αt→0, and the ϵ-loss weight αt2/σt2 Eq. (9) is zero.

  • Velocity prediction (§4). There are three fixes: (1) predict x, (2) predict x and ϵ together, or (3) predict v≡αtϵ−σtx, with x^=αtzt−σtv^θ(zt). All three work with distillation, and v is the most stable, but the experiments use (1) on CIFAR-10 and (2) on the other three datasets.

  • Truncated-SNR weighting (§4). Lθ=max(‖x−x^t‖22,‖ϵ−ϵ^t‖22)=max(αt2/σt2,1)‖x−x^t‖22, which is not zero at zero SNR.

def cosine_logsnr(u, logsnr_min=-20.0, logsnr_max=20.0):
    b = atan(exp(-0.5 * logsnr_max))
    a = atan(exp(-0.5 * logsnr_min)) - b
    return -2 * torch.log(torch.tan(a * u + b))  # alpha_t = cos(pi t / 2), clipped at |logsnr| = 20

def alpha_sigma(logsnr):
    return torch.sigmoid(logsnr).sqrt(), torch.sigmoid(-logsnr).sqrt()

def predict(model, z, logsnr, mean_type="x"):
    alpha, sigma = alpha_sigma(logsnr)
    out = model(z, logsnr)
    x = out if mean_type == "x" else alpha * z - sigma * out  # "v": x = alpha z - sigma v
    return x, (z - alpha * x) / sigma

def ddim_step(model, z_t, logsnr_t, logsnr_s):
    x, eps = predict(model, z_t, logsnr_t)
    alpha_s, sigma_s = alpha_sigma(logsnr_s)
    return alpha_s * x + sigma_s * eps, x  # Eq. 7

def train(student, teacher, x, num_steps):
    # 1. Sample a discrete time and add noise to the data.
    i = torch.randint(0, num_steps, (x.shape[0],))
    logsnr_t = cosine_logsnr((i + 1) / num_steps)
    alpha_t, sigma_t = alpha_sigma(logsnr_t)
    z_t = alpha_t * x + sigma_t * torch.randn_like(x)  # Eq. 1

    # 2. Run two teacher DDIM steps.
    logsnr_mid, logsnr_s = cosine_logsnr((i + 0.5) / num_steps), cosine_logsnr(i / num_steps)
    with no_grad():
        z_mid, _ = ddim_step(teacher, z_t, logsnr_t, logsnr_mid)
        z_s, x_mid = ddim_step(teacher, z_mid, logsnr_mid, logsnr_s)

    # 3. Invert one student DDIM step to get the target.
    alpha_s, sigma_s = alpha_sigma(logsnr_s)
    frac = sigma_s / sigma_t
    x_target = (z_s - frac * z_t) / (alpha_s - frac * alpha_t)  # never clipped
    x_target = where(i == 0, x_mid, x_target)  # final step: the teacher's own prediction
    eps_target = (z_t - alpha_t * x_target) / sigma_t

    # 4. Calculate the loss with truncated-SNR weighting.
    x_pred, eps_pred = predict(student, z_t, logsnr_t)
    x_mse = (x_pred - x_target).square().flatten(1).mean(1)
    eps_mse = (eps_pred - eps_target).square().flatten(1).mean(1)  # Eq. 9: SNR times x_mse
    loss = torch.maximum(x_mse, eps_mse).mean()  # max(1, SNR) weighting

def distill(model, data, start_num_steps=8192, end_num_steps=1, steps_per_iter=50000):
    teacher, num_steps = model, start_num_steps
    while num_steps > end_num_steps:
        num_steps //= 2
        student = copy(teacher)  # the student starts as the teacher
        for _ in range(steps_per_iter):
            train(student, teacher, next(data), num_steps)
        teacher = student
    return teacher

def sample(model, num_steps, shape):
    z = torch.randn(shape)
    for i in reversed(range(num_steps)):
        z, x = ddim_step(model, z, cosine_logsnr((i + 1) / num_steps), cosine_logsnr(i / num_steps))
    return x

def ddim_timesteps(num_steps=100, quadratic=True):
    if quadratic:
        return [int(s) for s in linspace(0, sqrt(800), num_steps) ** 2]  # CIFAR-10, CelebA
    return list(range(0, 1000, 1000 // num_steps))  # LSUN: 50 linear steps

def make_pair(teacher, bs, res, num_steps=100, quadratic=True, beta_min=1e-4, beta_max=0.02):
    alphas = torch.cumprod(1 - torch.linspace(beta_min, beta_max, 1000), dim=0)
    seq = ddim_timesteps(num_steps, quadratic)
    x_T = x = torch.randn(bs, res, res, 3)
    for i, j in zip(reversed(seq), reversed([-1] + seq[:-1])):
        a, a_next = alphas[i], (alphas[j] if j >= 0 else 1.0)
        eps = teacher(x, i)
        x = sqrt(a_next / a) * (x - sqrt(1 - a) * eps) + sqrt(1 - a_next) * eps  # Eq. 8
    return x_T, x.clamp(-1, 1)  # 1.024M pairs, made once and reused

def student(model, x_T, t_top=999):
    return x_T - model(x_T, t_top)  # the teacher's net and weights, fixed at the top timestep

def train(model, x_T, x_0, use_l2=True):
    # 1. Predict the image in one step.
    x_pred = student(model, x_T)

    # 2. Calculate the regression loss on the synthesized pair.
    err = x_0 - x_pred
    loss = (err.square() if use_l2 else err.abs()).flatten(1).sum(1).mean()  # Eq. 11 (CIFAR-10 uses L1)

def sample(model, bs, res):
    return student(model, torch.randn(bs, res, res, 3))
